#!/bin/bash
# shellcheck disable=SC2016

# 检查 OpenClaw Gateway 是否正在运行
# 如果未运行，依次尝试：
#   1. openclaw doctor --fix --non-interactive
#   2. 若仍未恢复，执行 run_diagnose.sh 深度修复
#
# 输出格式：
#   running  — Gateway 正在运行（端口已监听）
#   stopped  — 端口未监听，执行自动修复

GATEWAY_PORT=18789
DIAGNOSE_SCRIPT="/root/.kimi/kimi-claw/scripts/run_diagnose.sh"

check_port() {
    # 方法 1: ss
    if ss -tlnp 2>/dev/null | grep -q ":${GATEWAY_PORT} "; then
        return 0
    fi

    # 方法 2: /proc/net/tcp
    local port_hex
    port_hex=$(printf '%04X' "$GATEWAY_PORT")
    if grep -qi ":${port_hex} " /proc/net/tcp 2>/dev/null || \
       grep -qi ":${port_hex} " /proc/net/tcp6 2>/dev/null; then
        return 0
    fi

    # 方法 3: TCP 连接探测（兜底）
    if (: >/dev/tcp/127.0.0.1/$GATEWAY_PORT) 2>/dev/null; then
        return 0
    elif command -v nc >/dev/null 2>&1 && nc -z -w1 127.0.0.1 "$GATEWAY_PORT" >/dev/null 2>&1; then
        return 0
    elif command -v curl >/dev/null 2>&1; then
        curl -s --connect-timeout 1 --max-time 2 -o /dev/null "http://127.0.0.1:$GATEWAY_PORT/" >/dev/null 2>&1
        [ $? -ne 7 ] && return 0
    fi

    return 1
}

# ── 第一次检测 ──

if check_port; then
    echo "running"
    exit 0
fi

# ── 第一轮修复: openclaw doctor ──

echo "stopped"
echo "尝试 openclaw doctor --fix ..."
openclaw doctor --fix --non-interactive

sleep 5

# ── 第二次检测 ──

if check_port; then
    echo "running"
    exit 0
fi

# ── 第二轮修复: run_diagnose.sh ──

echo "doctor fix 后 Gateway 仍未恢复，执行 run_diagnose.sh ..."
if [ -x "$DIAGNOSE_SCRIPT" ]; then
    bash "$DIAGNOSE_SCRIPT"
else
    echo "错误: $DIAGNOSE_SCRIPT 不存在或不可执行" >&2
    exit 1
fi
